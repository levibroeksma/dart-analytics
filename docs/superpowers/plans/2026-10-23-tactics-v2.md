## Spec refinements (write back into the spec on approval)

1. **`recordMarksDart` reads the game status, not a parameter (§5).** Both marks states carry a game-level `status` (`CricketState`, `TacticsState`) that equals "the active seat is complete" for one seat. The function uses `before.status` / `after.status`; no `isEnded` parameter.
2. **Seat rota needs no change (§5, §11).** `activeSeat(…, "PER_SEAT")` with no completion predicate (V1's call) is pure alternation, so a closed-out seat keeps getting turns. A test pins it.
3. **`wouldComplete` clones the engine (§5).** `new TacticsEngine(config, facts(), key).record(obs).status === "COMPLETE"` inside a try replaces the V1 body for both seat counts: one path, no per-count branch.
4. **No `rowsRef` outside the shell (§7).** `SeatedPlayScreen` wraps the `rows` slot in `x-data="{ rowsRef: <seat ref expr> }"`; `MarkRows` takes `rowsExpr` (default `objectiveRows()`, Cricket unchanged).
5. **TacticsConfig is shared (§6).** V2 has no config fields, so `RULESET_CONFIGS.TACTICS_V2` is `TacticsConfig` itself and `tacticsValidator` serves both keys.

```ts
export type MarksTally = {
  fields: Pick<
    MarksSeatState,
    "marks" | "closedAtDart" | "objectiveHits" | "trebleHits" | "dartsThrown"
  >;
  spare: number;
};

export type MarksDuelSeat = MarksSeatState & { points: number };

export type MarksDuelSpec<TSeat extends MarksDuelSeat> = {
  initialSeat: (seat: SeatFact) => TSeat;
  hitOf: (seat: TSeat, opponent: TSeat, observation: DartObservation) => MarksHit;
  valueOf: (objectiveIndex: number) => number | null;
};

export type MarksDuelFold<TSeat extends MarksDuelSeat> = {
  seats: TSeat[];
  winnerParticipantRef: string | null;
};
```

  (`SeatFact` is imported in `types.ts` from `@lib/types` if the file does not yet import it.)
- Consumes: `seatOf` (`seat-rota.module.ts`), `refuseWhenEnded` (`turn-log.module.ts`).
- Semantics: a dart's hit scores `spare × valueOf(index)` only when `valueOf` is non-null and the opponent's marks on that objective are below 3. A seat is closed out when every mark is 3. The first dart that leaves the throwing seat closed out with `points >= opponent.points` sets the winner, zeroes that seat's `dartsThisVisit`, and any later dart throws `game has ended`. A closed-out seat that has not won keeps throwing.

```ts
const spec: MarksDuelSpec<MarksDuelSeat> = {
  initialSeat: (seat) => ({ ...initialMarksSeat(seat, 3), points: 0 }),

    intendedZoneKey: null,
    hitTargetNumber: number,
    hitZoneKey: zone,
    score: 0,
    locationX: null,
    locationY: null,

    sequence: key,
```

```ts
export function tallyMarksDart(state: MarksSeatState, hit: MarksHit): MarksTally {
  const dartsThrown = state.dartsThrown + 1;
  const marks = [...state.marks];
  const closedAtDart = [...state.closedAtDart];
  const objectiveHits = [...state.objectiveHits];
  let spare = 0;
  if (hit) {
    const index = hit.objectiveIndex;
    const total = marks[index] + hit.marks;
    objectiveHits[index] += 1;
    marks[index] = Math.min(MARKS_TO_CLOSE, total);
    spare = Math.max(0, total - MARKS_TO_CLOSE);
    if (state.marks[index] < MARKS_TO_CLOSE && marks[index] === MARKS_TO_CLOSE) {
      closedAtDart[index] = dartsThrown;
    }
  }
  return {
    fields: {
      marks,
      closedAtDart,
      objectiveHits,
      trebleHits: state.trebleHits + (hit?.marks === 3 ? 1 : 0),
      dartsThrown,
    },
    spare,
  };
}

export function applyMarksDart(state: MarksSeatState, hit: MarksHit): MarksSeatState {
  refuseWhenEnded(state.status !== "IN_PROGRESS", "game has ended");
  const { fields } = tallyMarksDart(state, hit);
  const complete = fields.marks.every((count) => count === MARKS_TO_CLOSE);
  const visitDarts = state.dartsThisVisit + 1;
  return {
    ...state,
    ...fields,
    dartsThisVisit: complete || visitDarts === DARTS_PER_VISIT ? 0 : visitDarts,
    status: complete ? "COMPLETE" : "IN_PROGRESS",
  };
}
```

```ts
import {
  DARTS_PER_VISIT,
  MARKS_TO_CLOSE,
  tallyMarksDart,
} from "./marks-close.module";
import { seatOf } from "./seat-rota.module";
import { toObservation } from "./seat-state.module";
import { refuseWhenEnded } from "./turn-log.module";
import type { SeatFact } from "@lib/types";
import type {
  MarksDuelFold,
  MarksDuelSeat,
  MarksDuelSpec,
  MarksHit,
  TurnFact,
} from "./types";

function applyDuelDart<TSeat extends MarksDuelSeat>(
  seat: TSeat,
  opponent: TSeat,
  hit: MarksHit,
  valueOf: (objectiveIndex: number) => number | null,
): TSeat {
  const { fields, spare } = tallyMarksDart(seat, hit);
  const value =
    hit && opponent.marks[hit.objectiveIndex] < MARKS_TO_CLOSE
      ? valueOf(hit.objectiveIndex)
      : null;
  const visitDarts = seat.dartsThisVisit + 1;
  return {
    ...seat,
    ...fields,
    points: seat.points + (value === null ? 0 : spare * value),
    dartsThisVisit: visitDarts === DARTS_PER_VISIT ? 0 : visitDarts,
    status: fields.marks.every((count) => count === MARKS_TO_CLOSE)
      ? "COMPLETE"
      : "IN_PROGRESS",
  };
}

/**
 * Walks both seats' darts in log order, because a dart's points depend on the
 * opponent's marks at that dart. A dart that leaves its seat closed out with
 * points at least equal to the opponent's wins the game on the spot.
 * @throws when a dart follows the winning one.
 */
export function foldMarksDuel<TSeat extends MarksDuelSeat>(
  turns: readonly TurnFact[],
  seats: readonly [SeatFact, SeatFact],
  spec: MarksDuelSpec<TSeat>,
): MarksDuelFold<TSeat> {
  const states = seats.map(spec.initialSeat);
  let winnerParticipantRef: string | null = null;
  for (const turn of turns) {
    const index = seats.indexOf(seatOf(turn, seats));
    for (const dart of turn.darts) {
      refuseWhenEnded(winnerParticipantRef !== null, "game has ended");
      const opponent = states[1 - index];
      const next = applyDuelDart(
        states[index],
        opponent,
        spec.hitOf(states[index], opponent, toObservation(dart)),
        spec.valueOf,
      );
      states[index] = next;
      if (next.status === "COMPLETE" && next.points >= opponent.points) {
        states[index] = { ...next, dartsThisVisit: 0 };
        winnerParticipantRef = next.participantRef;
      }
    }
  }
  return { seats: states, winnerParticipantRef };
}
```

```ts
const NINE = TACTICS_OBJECTIVES.length;

function seatByRef(state: TacticsState | null, ref: string | undefined) {
  return state?.seats.find((seat) => seat.participantRef === ref);
}

function rowsOf(state: TacticsState | null, ref: string | undefined): TacticsObjectiveRow[] {
  const seat = seatByRef(state, ref);
  const opponent = state?.seats.find((candidate) => candidate.participantRef !== ref);
  const duel = (state?.seats.length ?? 0) > 1;
  return TACTICS_OBJECTIVES.map((objective, i) => {
    const marks = seat?.marks[i] ?? 0;
    const opponentMarks = duel ? (opponent?.marks[i] ?? 0) : null;
    return {
      label: objectiveLabel(objective),
      objective,
      marks,
      closed: marks === 3,
      opponentMarks,
      scoring: opponentMarks !== null && marks === 3 && opponentMarks < 3,
      dead: opponentMarks !== null && marks === 3 && opponentMarks === 3,
    };
  });
}
```
